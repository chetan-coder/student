## Problem 3:

let a = 9;
let b = 4;
let c = 3;

let addAB = a + b; // 9 + 4 = 13

let multipliedResult = addAB * c; // 13 * 3 = 39

let remainder = multipliedResult % b; // 39 % 4 = 3

console.log("a:", a);
console.log("b:", b);
console.log("c:", c);
console.log("a + b =", addAB);
console.log("(a + b) * c =", multipliedResult);
console.log("Final result: ((a + b) * c) % b =", remainder);
