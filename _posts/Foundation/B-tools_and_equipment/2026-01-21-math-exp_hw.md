## Problem 1:

// Subtract 4 from num

## Problem 3:

// A slightly more complex expression
// Step 1: Create three variables and give them values
let a = 9;
let b = 4;
let c = 3;
// Step 2: Add a and b // This will calculate (a + b)
let addAB = a + b; // 9 + 4 = 13
// Step 3: Multiply the result by c // This will calculate (a + b) * c
let multipliedResult = addAB * c; // 13 * 3 = 39
// Step 4: Find the remainder when that result is divided by b
// This will calculate ((a + b) * c) % b
let remainder = multipliedResult % b; // 39 % 4 = 3
// Step 5: Print all the steps and the final result
console.log("a:", a);
console.log("b:", b);
console.log("c:", c);
console.log("a + b =", addAB);
console.log("(a + b) * c =", multipliedResult);
console.log("Final result: ((a + b) * c) % b =", remainder);
